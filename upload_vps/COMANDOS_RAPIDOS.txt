🐳 DOCKER - GERENCIAR CONTAINERS
--------------------------------------------------------------------------------
cd /var/www/i9script

# Ver containers rodando
docker ps

# Ver todos os containers (incluindo parados)
docker ps -a

# Iniciar containers
docker-compose -f docker-compose.production.yml up -d

# Parar containers
docker-compose -f docker-compose.production.yml down

# Reiniciar containers
docker-compose -f docker-compose.production.yml restart

# Ou usar script:
bash scripts/restart.sh

# Ver status dos serviços
docker-compose -f docker-compose.production.yml ps


📋 LOGS
--------------------------------------------------------------------------------
# Ver todos os logs
bash scripts/logs.sh

bash scripts/logs.sh backend

# Logs do MySQL
bash scripts/logs.sh mysql

# Logs direto do Docker
docker logs i9script-backend-prod
docker logs i9script-mysql-prod

# Logs em tempo real
docker logs -f i9script-backend-prod


💾 MYSQL - BANCO DE DADOS
--------------------------------------------------------------------------------
# Acessar MySQL via Docker
docker exec -it i9script-mysql-prod mysql -ui9script -p
# Senha: (a que você definiu no .env)

# Dentro do MySQL:
USE i9script_db;
SHOW TABLES;
SELECT * FROM users;
EXIT;

# Backup do banco
docker exec i9script-mysql-prod mysqldump -ui9script -p i9script_db > backup_$(date +%Y%m%d).sql

# Restaurar backup
docker exec -i i9script-mysql-prod mysql -ui9script -p i9script_db < backup.sql

# Ver tamanho do banco
docker exec i9script-mysql-prod mysql -ui9script -p -e "SELECT table_schema AS 'Database', ROUND(SUM(data_length + index_length) / 1024 / 1024, 2) AS 'Size (MB)' FROM information_schema.TABLES WHERE table_schema='i9script_db' GROUP BY table_schema;"

🔥 FIREWALL
--------------------------------------------------------------------------------

# Permitir porta

# Bloquear porta

# Habilitar firewall

# Desabilitar firewall
